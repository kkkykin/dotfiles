#+name: gen-include
#+begin_src elisp
(if-let* ((configs (directory-files "." t "^_.+\\.conf")))
    (cl-loop
     for conf in configs
     for base = (file-name-base conf)
     if (string-prefix-p "__" base)
     collect (format "[includeIf \"gitdir:~/%s/\"]\n\tpath = ~/%s"
                     (substring base 2)
                     (substring base 2))
     into gitdirs
     else
     collect (format "path = %s" conf)
     into include
     finally return
     (string-join
      (append
       (when include
         (list (format "[include]\n\t%s"
                       (string-join include "\n\t"))))
       gitdirs)
      "\n\n"))
  "")
#+end_src
